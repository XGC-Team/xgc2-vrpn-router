cmake_minimum_required(VERSION 3.16)
project(xgc2_vrpn_router VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(Threads REQUIRED)
find_package(XgcXrpc REQUIRED CONFIG COMPONENTS http bootstrap diagnostics)
find_package(jsoncpp REQUIRED CONFIG)
find_package(c-ares 1.34.8 REQUIRED CONFIG)
include(CTest)

set(VRPN_ROOT "" CACHE PATH "Optional VRPN install prefix")
find_path(VRPN_INCLUDE_DIR vrpn_Tracker.h HINTS "${VRPN_ROOT}/include")
find_library(VRPN_SERVER_LIBRARY NAMES vrpnserver HINTS "${VRPN_ROOT}/lib" "${VRPN_ROOT}/lib64")
find_library(QUAT_LIBRARY NAMES quat HINTS "${VRPN_ROOT}/lib" "${VRPN_ROOT}/lib64")
find_library(LIBUSB_LIBRARY NAMES usb-1.0)

if(NOT VRPN_INCLUDE_DIR OR NOT VRPN_SERVER_LIBRARY OR NOT LIBUSB_LIBRARY)
  message(FATAL_ERROR "VRPN server headers/library or libusb-1.0 not found. Set VRPN_ROOT to a VRPN install prefix.")
endif()

add_library(router-domain STATIC src/router.cpp src/config.cpp src/resolver.cpp)
target_include_directories(router-domain PUBLIC src ${VRPN_INCLUDE_DIR})
target_link_libraries(router-domain PUBLIC ${VRPN_SERVER_LIBRARY} Threads::Threads JsonCpp::JsonCpp c-ares::cares)
if(QUAT_LIBRARY)
  target_link_libraries(router-domain PUBLIC ${QUAT_LIBRARY})
endif()
target_link_libraries(router-domain PUBLIC ${LIBUSB_LIBRARY})
add_library(router-service STATIC src/service.cpp src/application.cpp src/runtime.cpp)
target_link_libraries(router-service PUBLIC router-domain XgcXrpc::http XgcXrpc::bootstrap XgcXrpc::diagnostics)
foreach(target router-domain router-service)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()

add_executable(xgc2-vrpn-router src/main.cpp)
target_link_libraries(xgc2-vrpn-router PRIVATE router-service)
if(BUILD_TESTING)
  add_executable(router-config-test tests/config_test.cpp)
  target_link_libraries(router-config-test PRIVATE router-domain)
  target_compile_options(router-config-test PRIVATE -UNDEBUG)
  add_test(NAME domain-configuration-boundaries COMMAND router-config-test)
  add_executable(router-resolver-test tests/resolver_test.cpp)
  target_link_libraries(router-resolver-test PRIVATE router-domain)
  target_compile_options(router-resolver-test PRIVATE -UNDEBUG)
  add_test(NAME configuration-dns-deadline-cancellation COMMAND router-resolver-test)
  add_executable(router-native-peer tests/native_peer.cpp)
  target_link_libraries(router-native-peer PRIVATE router-domain)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME native-process-management COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/process_test.py
    --host $<TARGET_FILE:xgc2-vrpn-router> --peer $<TARGET_FILE:router-native-peer>)
  set_tests_properties(native-process-management PROPERTIES TIMEOUT 45)
  set(VRPN_INTEROP_ROOT "" CACHE PATH "Optional pristine official VRPN prefix for wire interoperability tests")
  if(VRPN_INTEROP_ROOT)
    add_executable(router-official-peer tests/native_peer.cpp)
    target_include_directories(router-official-peer PRIVATE "${VRPN_INTEROP_ROOT}/include")
    target_link_libraries(router-official-peer PRIVATE "${VRPN_INTEROP_ROOT}/lib/libvrpnserver.a"
      "${VRPN_INTEROP_ROOT}/lib/libquat.a" ${LIBUSB_LIBRARY} Threads::Threads)
    add_test(NAME official-native-wire-interoperability COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/process_test.py
      --host $<TARGET_FILE:xgc2-vrpn-router> --peer $<TARGET_FILE:router-official-peer>)
    set_tests_properties(official-native-wire-interoperability PROPERTIES TIMEOUT 50)
  endif()
endif()

install(TARGETS xgc2-vrpn-router RUNTIME DESTINATION bin)
install(FILES config/router.json DESTINATION share/xgc2-vrpn-router)
install(FILES systemd/xgc2-vrpn-router.service DESTINATION /lib/systemd/system)
